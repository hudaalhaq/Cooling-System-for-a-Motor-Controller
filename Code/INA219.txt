#include <Wire.h>
#include <INA219_WE.h>
#define I2C_ADDRESS 0x40

INA219_WE ina219 = INA219_WE(I2C_ADDRESS);

// variabel untuk menampung data tegangan, arus dan daya
float teganganBeban = 0.0;
float arus_mA = 0.0;
float daya_mW = 0.0; 
  
void setup() {
  Serial.begin(9600);
  Wire.begin();
  if(!ina219.init()){
    Serial.println("INA219 TIDAK TEHUBUNG!");
  }
}

void loop() {
  arus_mA = ina219.getCurrent_mA();
  daya_mW = ina219.getBusPower();
  teganganBeban  = ina219.getBusVoltage_V() + (ina219.getShuntVoltage_mV()/1000); // ambil data sensor INA219  
  Serial.print("Tegangan Beban [V]: ");
  Serial.println(teganganBeban);
  Serial.print("Arus[mA]: ");
  Serial.println(arus_mA);
  Serial.print("Daya [mW]: ");
  Serial.println(daya_mW);
}
